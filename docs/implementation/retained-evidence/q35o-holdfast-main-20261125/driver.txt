def sha(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()

def identity(path):
    return {"bytes": path.stat().st_size, "sha256": sha(path)}

def write(name, value):
    (out / name).write_text(json.dumps(value, sort_keys=True, indent=2) + "\n")

def library_inputs():
    return {
        str(root): {str(p.relative_to(root)): sha(p) for p in sorted(root.rglob("*"))
                    if p.is_file() and p.suffix in {".v", ".vo", ".cmxs", ".cma", ".cmxa"}}
        for root in (corelib, stdlib, stdpp, support)
    }

def limits():
    resource.setrlimit(resource.RLIMIT_FSIZE, (256 * 1024**2, 256 * 1024**2))

def run(mode, phase, args, *, silent=False):
    cwd = out / mode
    stdout = cwd / (phase + ".stdout")
    stderr = cwd / (phase + ".stderr")
    argv = [str(rocq), "c", "-q", "-boot", "-R", str(corelib), "Corelib",
            "-Q", str(stdlib), "Stdlib", "-Q", str(stdpp), "stdpp",
            "-R", str(support), "SailStdpp", "-R", str(cwd), "", *map(str, args)]
    at = time.monotonic()
    print(mode, phase, flush=True)
    result = {"mode": mode, "phase": phase, "argv": argv, "cwd": str(cwd)}
    records.append(result)
    try:
        with stdout.open("wb") as output, stderr.open("wb") as errors:
            result["exit"] = subprocess.run(argv, cwd=cwd, stdout=output, stderr=errors,
                                            timeout=240, preexec_fn=limits).returncode
    finally:
        result.update(elapsed_seconds=round(time.monotonic() - at, 3),
                      stdout=identity(stdout), stderr=identity(stderr))
        write("commands.json", records)
    if result["exit"] != 0 or (silent and stderr.stat().st_size):
        last = stderr.read_bytes()[-5000:].decode(errors="replace")
        (out / "stop-diagnostic.txt").write_text(last)
        raise ValueError(f"{mode} {phase}: exit {result['exit']}\n{last}")
    return stdout

def query(mode, phase, module, body):
    filename = out / mode / (proofreading.query_module(phase, module) + ".v")
    filename.write_text(body)
    return run(mode, phase + "_" + module, [filename], silent=True).read_text()

def read_module(mode, module):
    names = [item["name"] for item in
             proofaudit.inventory(query(mode, "Inventory", module, proofaudit.inventory_query(module)), module)]
    if not names:
        raise ValueError("native inventory is empty")
    facts_name = proofreading.query_module("Facts", module)
    facts = proofreading.answers(
        query(mode, "Facts", module, proofreading.facts_query(module, names)),
        [f"{name}|{kind}" for name in names for kind in ("check", "about")],
    )
    axioms = {f"{module}.{name}" for name in generations[mode]["axioms"]["names"]}
    if len(axioms) != 31 or not axioms <= set(names):
        raise ValueError(f"{mode}: native inventory does not contain the expected 31 emitted axioms")
    abouts = {name: proofreading.parse_about(
        name, proofreading.fresh_universes(facts[f"{name}|about"], facts_name))
        for name in names if name not in axioms}
    printed = [name for name in names if name in axioms or abouts[name].printed]
    bodies_name = proofreading.query_module("Bodies", module)
    bodies = proofreading.answers(
        query(mode, "Bodies", module, proofreading.bodies_query(module, printed)),
        [f"{name}|print" for name in printed],
    )
    entries = {}
    for name in names:
        check = proofreading.fresh_universes(facts[f"{name}|check"], facts_name)
        body = proofreading.fresh_universes(bodies[f"{name}|print"], bodies_name) if f"{name}|print" in bodies else None
        if name in axioms:
            about = proofreading.fresh_universes(facts[f"{name}|about"], facts_name)
            if f"Expands to: Constant {name}" not in about.splitlines() or body is None or not body.startswith("*** ["):
                raise ValueError(f"{mode}: expected emitted axiom {name!r}: {about!r}; {body!r}")
            entries[name] = {"kind": "Axiom", "opacity": "no body", "universes": "see About",
                             "check": check, "about": "\n".join(line for line in about.splitlines()
                                                                 if not line.startswith("Declared in ")),
                             "print": body}
        else:
            entries[name] = proofreading.entry(check, abouts[name], body)
    frozen = out / mode / (module + ".frozen.json")
    frozen.write_text(json.dumps(entries, sort_keys=True, indent=2) + "\n")
    comparison[mode] = {"declarations": len(names), "print_targets": len(printed),
                        "frozen": identity(frozen)}
    return entries

        })
