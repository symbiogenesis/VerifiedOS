def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def query(mode,module,phase,text):
    name=proofreading.query_module(phase,module); f=T/(name+'.v'); f.write_text(text,newline=''); argv=[str(P/'bin/rocq'),'c','-q',*BASE,'-R',str(L/('probe-'+mode)),'',*STRICT,str(f)]

    records.append(dict(mode=mode,module=module,phase=phase,argv=argv,cwd=str(T),exit=d.returncode,elapsed=time.monotonic()-begin,stdout=str(out),stderr=str(err),stdout_sha256=sha(out),stderr_sha256=sha(err)))
    (L/'frozen-receipt.json').write_text(json.dumps(records,indent=2)+'\n')
    assert d.returncode==0 and not d.stderr,(phase,d.stderr.decode())
    assert len(d.stdout)<64*1024*1024
    return d.stdout.decode()
